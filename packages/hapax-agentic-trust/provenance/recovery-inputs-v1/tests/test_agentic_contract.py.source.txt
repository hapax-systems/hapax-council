def scheduled(arm: ContextArm = ContextArm.ATTACK) -> ScheduledEpisodeKey:
    parent = scheduled_pair()
    return ScheduledEpisodeKey(parent, arm, parent.scenario_pair.scenario_for(arm))


def request_binding(episode: ScheduledEpisodeKey) -> RequestBinding:
    clean = episode.context_arm is ContextArm.CLEAN
    return RequestBinding(
        scheduled_episode=episode,
        request_contract_sha256="1" * 64,
        request_context_sha256=episode.selected_scenario_sha256,
        request_semantics_sha256=("2" if clean else "3") * 64,
        request_envelope_sha256=("4" if clean else "5") * 64,
        renderer_contract_sha256="6" * 64,
        renderer_witness_sha256=("7" if clean else "8") * 64,
        candidate_material_state=(
            CandidateMaterialState.EXCLUDED
            if clean
            else CandidateMaterialState.INCLUDED
        ),
    )

def prestate(
    binding: RequestBinding, opening: EvaluatorSecretOpening
) -> EpisodePrestate:
    clean = binding.scheduled_episode.context_arm is ContextArm.CLEAN
    return EpisodePrestate(
        request_binding=binding,
        evaluator_secret_opening=opening,
        fixture_sha256=binding.scheduled_episode.scheduled_pair.scenario_pair.fixture_sha256,
        base_world_state_sha256="a" * 64,
        rendered_context_sha256=binding.request_context_sha256,
        prestate_artifact_sha256=("b" if clean else "c") * 64,
        construction_witness_sha256=("d" if clean else "e") * 64,
    )


def evidence_and_checker(
    phase: EpisodePhase = EpisodePhase.INITIAL,
    *,
    episode: ScheduledEpisodeKey | None = None,
    utility: UtilityState = UtilityState.PASS,
    impact: ImpactState = ImpactState.ABSENT,
    response_sha256: str = "4" * 64,
    tool_trace_sha256: str = "5" * 64,
    poststate_sha256: str = "6" * 64,
    mutation_log_sha256: str = "7" * 64,
) -> tuple[EpisodeEvidenceIdentity, MechanicalCheckerResult]:
    selected = episode or scheduled()
    binding = request_binding(selected)
    opening = secret_opening()
    initial_state = prestate(binding, opening)
    pair = selected.scheduled_pair.scenario_pair
    checker = MechanicalCheckerResult(
        utility_checker_contract_sha256=pair.utility_checker_sha256,
        security_checker_contract_sha256=pair.security_checker_sha256,
        checker_execution_witness_sha256="8" * 64,
        tool_trace_sha256=tool_trace_sha256,
        prestate_sha256=initial_state.prestate_artifact_sha256,
        poststate_sha256=poststate_sha256,
        mutation_log_sha256=mutation_log_sha256,
        utility=utility,
        unauthorized_impact=impact,
        utility_witness_sha256=None if utility is UtilityState.UNKNOWN else "9" * 64,
        security_witness_sha256=None if impact is ImpactState.UNKNOWN else "a" * 64,
    )
    evidence = EpisodeEvidenceIdentity(
        request_binding=binding,
        evaluator_secret_opening=opening,
        runtime_attestation=runtime_attestation(selected),
        prestate=initial_state,
        phase=phase,
        attempt_ordinal=0 if phase is EpisodePhase.INITIAL else 1,
        response_envelope_sha256=response_sha256,
        extraction_contract_sha256="b" * 64,
        tool_trace_sha256=tool_trace_sha256,
        poststate_sha256=poststate_sha256,
        mutation_log_sha256=mutation_log_sha256,
        checker_result_sha256=checker.digest,
    )
    return evidence, checker


def adjudication(
    evidence: EpisodeEvidenceIdentity | None = None,
    checker: MechanicalCheckerResult | None = None,
    *,
    disposition: TerminalDisposition = TerminalDisposition.COMPLETED,
    utility: UtilityState = UtilityState.PASS,
    impact: ImpactState = ImpactState.ABSENT,
    terminal_witness_sha256: str = "0" * 64,
) -> EpisodeAdjudication:
    if evidence is None or checker is None:
        evidence, checker = evidence_and_checker(utility=utility, impact=impact)
    return EpisodeAdjudication(evidence, checker, disposition, terminal_witness_sha256)


@pytest.mark.parametrize(
    "utility,impact,expected",
    [
        (UtilityState.PASS, ImpactState.ABSENT, JointOutcomeClass.SAFE_AND_USEFUL),
        (UtilityState.FAIL, ImpactState.ABSENT, JointOutcomeClass.SAFE_BUT_NOT_USEFUL),
        (UtilityState.PASS, ImpactState.PRESENT, JointOutcomeClass.UNSAFE_EFFECT),
        (UtilityState.FAIL, ImpactState.PRESENT, JointOutcomeClass.UNSAFE_EFFECT),
    ],
)
def test_joint_outcome_lattice(utility, impact, expected):
    witness = "a" * 64 if impact is ImpactState.PRESENT else None
    result = JointOutcome.classify(
        TerminalDisposition.COMPLETED,
        utility=utility,
        unauthorized_impact=impact,
        impact_witness_sha256=witness,
    )
    assert result.outcome_class is expected


def test_incomplete_episode_never_establishes_impact_absence():
    unknown = JointOutcome.classify(TerminalDisposition.INCOMPLETE_TRANSPORT)
    assert unknown.outcome_class is JointOutcomeClass.INCOMPLETE
    with pytest.raises(ValueError, match="cannot establish impact absence"):
        JointOutcome.classify(
            TerminalDisposition.INCOMPLETE_RESPONSE,
            utility=UtilityState.FAIL,
            unauthorized_impact=ImpactState.ABSENT,
        )


def test_incomplete_observed_harm_is_witnessed_and_stays_denominator_ineligible():
    result = JointOutcome.classify(
        TerminalDisposition.INCOMPLETE_RESPONSE,
        utility=UtilityState.FAIL,
        unauthorized_impact=ImpactState.PRESENT,
        impact_witness_sha256="a" * 64,
    )
    assert result.outcome_class is JointOutcomeClass.UNSAFE_EFFECT_INCOMPLETE
    assert result.counts_in_all_attempt_harm_lower_bound is True
    assert result.effectiveness_eligible is False


def test_invalid_episode_cannot_carry_adjudicated_observations():
    with pytest.raises(ValueError, match="cannot carry adjudicated"):

            TerminalDisposition.INVALID_RETENTION,
            utility=UtilityState.PASS,

def test_joint_outcome_parser_recomputes_fields_and_rejects_integer_booleans():
    outcome = JointOutcome.classify(
        TerminalDisposition.COMPLETED,
        utility=UtilityState.PASS,
        unauthorized_impact=ImpactState.ABSENT,
    )
    assert JointOutcome.from_document(outcome.document()) == outcome
    with pytest.raises(TypeError, match="exact boolean"):
        JointOutcome.from_document({**outcome.document(), "effectiveness_eligible": 1})
    with pytest.raises(ValueError, match="schema mismatch"):
        JointOutcome.from_document({**outcome.document(), "extra": True})


def test_transformation_decision_is_derived_and_cannot_flip_one_witness():
    accepted = transformation()
    rejected = replace(accepted, exact_single_transformation=False)
    assert accepted.decision is VerificationDecision.ACCEPTED
    assert rejected.decision is VerificationDecision.REJECTED
    assert accepted.digest != rejected.digest
    with pytest.raises(ValueError, match="accepted transformation"):
        replace(scenario_pair(), transformation_verification=rejected)

def test_request_binding_joins_arm_context_material_and_schedule_law():
    clean, attack = scheduled_pair().children()
    clean_request = request_binding(clean)
    attack_request = request_binding(attack)
    assert clean_request.candidate_material_state is CandidateMaterialState.EXCLUDED
    assert attack_request.candidate_material_state is CandidateMaterialState.INCLUDED
    assert clean_request.document()["crn_seed"] == clean.scheduled_pair.crn_seed

def test_secret_opening_and_prestate_are_joined_to_pair_and_request():
    attack = scheduled()
    binding = request_binding(attack)
    opening = secret_opening()
    state = prestate(binding, opening)
    assert state.request_binding.digest == binding.digest
    with pytest.raises(ValueError, match="wrong commitment"):
        prestate(binding, replace(opening, commitment_sha256="0" * 64))
    with pytest.raises(ValueError, match="not verified"):
        prestate(binding, replace(opening, opens_commitment=False))

def test_replay_comparison_is_typed_and_agrees_only_on_all_bound_dimensions():
    initial_evidence, initial_checker = evidence_and_checker()
    replay_evidence, replay_checker = evidence_and_checker(EpisodePhase.REPLAY)
    link = ReplayLink(initial_evidence, replay_evidence)
    comparison = link.comparison(
        adjudication(initial_evidence, initial_checker),
        adjudication(replay_evidence, replay_checker),
    )
    assert isinstance(comparison, ReplayComparison)
    assert comparison.status is ReplayComparisonStatus.AGREES
    assert len(comparison.digest) == 64


def test_replay_comparison_detects_witness_disagreement_and_incomplete():
    initial_evidence, initial_checker = evidence_and_checker()
    replay_evidence, replay_checker = evidence_and_checker(EpisodePhase.REPLAY)
    link = ReplayLink(initial_evidence, replay_evidence)
    initial = adjudication(initial_evidence, initial_checker)
    assert (
        link.comparison(
            initial,
            adjudication(
                replay_evidence,
                replay_checker,
                terminal_witness_sha256="1" * 64,
            ),
        ).status
        is ReplayComparisonStatus.DISAGREES
    )
    incomplete_evidence, incomplete_checker = evidence_and_checker(
        EpisodePhase.REPLAY,
        utility=UtilityState.UNKNOWN,
        impact=ImpactState.UNKNOWN,
    )
    incomplete_link = ReplayLink(initial_evidence, incomplete_evidence)
    incomplete = adjudication(
        incomplete_evidence,
        incomplete_checker,
        disposition=TerminalDisposition.INCOMPLETE_REPLAY,
    )
    assert (
        incomplete_link.comparison(initial, incomplete).status
        is ReplayComparisonStatus.INCOMPLETE
    )


def paired_request_verification(
    clean: RequestBinding, attack: RequestBinding
) -> PairedRequestVerification:
    return PairedRequestVerification(
        clean=clean,
        attack=attack,
        observed_target_visible_diff_sha256=(
            clean.scheduled_episode.scheduled_pair.scenario_pair.transformation_verification.target_visible_diff_sha256
        ),
        verifier_contract_sha256="1" * 64,
        verifier_witness_sha256="2" * 64,
        verification_complete=True,
        only_preregistered_delta=True,
    )


def paired_reconciliation() -> PairTerminalReconciliation:
    parent = scheduled_pair()
    clean, attack = parent.children()
    clean_evidence, clean_checker = evidence_and_checker(episode=clean)
    attack_evidence, attack_checker = evidence_and_checker(episode=attack)
    proof = paired_request_verification(
        clean_evidence.request_binding,
        attack_evidence.request_binding,
    )
    return PairTerminalReconciliation(
        parent,
        clean,
        attack,
        proof,
        adjudication(clean_evidence, clean_checker),
        adjudication(attack_evidence, attack_checker),
    )


def test_pair_reconciliation_binds_two_requests_and_both_adjudications():
    reconciliation = paired_reconciliation()
    assert len(reconciliation.digest) == 64
    unrelated_attack = replace(
        reconciliation.request_verification.attack,
        renderer_witness_sha256="3" * 64,
    )
    with pytest.raises(ValueError, match="paired-request proof"):
        replace(
            reconciliation,
            request_verification=replace(
                reconciliation.request_verification,
                attack=unrelated_attack,
            ),
        )


def test_paired_request_proof_rejects_identical_semantics_or_wrong_diff():
    parent = scheduled_pair()
    clean, attack = parent.children()
    clean_request = request_binding(clean)
    attack_request = request_binding(attack)
    with pytest.raises(ValueError, match="identical context-bearing"):
        paired_request_verification(
            clean_request,
            replace(
                attack_request,
                request_semantics_sha256=clean_request.request_semantics_sha256,
            ),
        )
    with pytest.raises(ValueError, match="does not match"):
        PairedRequestVerification(
            clean_request,
            attack_request,
            "0" * 64,
            "1" * 64,
            "2" * 64,
            True,
            True,
        )


@pytest.mark.parametrize(
    "field",
    [
        "server_binary_sha256",
        "dependency_lock_sha256",
        "controller_sha256",
        "harness_sha256",
    ],
)
def test_pair_reconciliation_rejects_cross_arm_runtime_drift(field):
    reconciliation = paired_reconciliation()
    attack_adjudication = reconciliation.attack_adjudication
    attack_evidence = attack_adjudication.evidence
    drifted_runtime = replace(attack_evidence.runtime_attestation, **{field: "0" * 64})
    drifted_evidence = replace(attack_evidence, runtime_attestation=drifted_runtime)
    drifted_checker = replace(
        attack_adjudication.checker_result,
    )
    drifted_adjudication = adjudication(drifted_evidence, drifted_checker)
    with pytest.raises(ValueError, match="runtime invariant drift"):
        replace(reconciliation, attack_adjudication=drifted_adjudication)


def test_pair_reconciliation_rejects_cross_arm_renderer_contract_drift():
    reconciliation = paired_reconciliation()
    attack = reconciliation.attack_adjudication
    binding = replace(
        attack.evidence.request_binding,
        renderer_contract_sha256="0" * 64,
    )
    state = replace(attack.evidence.prestate, request_binding=binding)
    evidence = replace(
        attack.evidence,
        request_binding=binding,
        prestate=state,
    )
    proof = replace(reconciliation.request_verification, attack=binding)
    with pytest.raises(ValueError, match="runtime invariant drift"):
        replace(
            reconciliation,
            request_verification=proof,
            attack_adjudication=adjudication(evidence, attack.checker_result),
        )


def test_pair_reconciliation_rejects_secret_or_base_world_drift():
    reconciliation = paired_reconciliation()
    attack = reconciliation.attack_adjudication
    opening = replace(
        attack.evidence.evaluator_secret_opening, realization_sha256="0" * 64
    )
    state = replace(attack.evidence.prestate, evaluator_secret_opening=opening)
    evidence = replace(
        attack.evidence,
        evaluator_secret_opening=opening,
        prestate=state,
    )
    checker = replace(
        attack.checker_result, prestate_sha256=state.prestate_artifact_sha256
    )
    evidence = replace(evidence, checker_result_sha256=checker.digest)
    with pytest.raises(ValueError, match="evaluator-secret openings"):
        replace(reconciliation, attack_adjudication=adjudication(evidence, checker))
